% A list is not cut by a floated exhibit. Each pass records the pages every list starts
% and ends on, and the page and height of every float. A list that ran across a page

% when it fits in the room that page has left; the decision is recorded, so the layout
% settles. A list too tall for that is reported for the author to fix. (A page here has no stretch, so only a

% Sets \Folio@room to the least room on the float pages the list ran onto (or \maxdimen
% when it ran onto none) and \Folio@cut to the first such page.
\newcommand{\Folio@listcheck}{%
  \Folio@room=\maxdimen \Folio@cut=0
  \@ifundefined{Folio@Ls@\theFolioListNo}{}{\@ifundefined{Folio@Le@\theFolioListNo}{}{%
    \Folio@q=\csname Folio@Ls@\theFolioListNo\endcsname\relax
    \loop\ifnum\Folio@q<\csname Folio@Le@\theFolioListNo\endcsname\relax
      \advance\Folio@q 1
      \@ifundefined{Folio@fp@\the\Folio@q}{}{%
        \ifnum\Folio@cut=0 \Folio@cut=\Folio@q \fi
        \dimen@=\dimexpr\textheight-\csname Folio@fp@\the\Folio@q\endcsname
          -\textfloatsep-4\baselineskip\relax
        \ifdim\dimen@<\Folio@room \Folio@room=\dimen@ \fi}%
    \repeat}}}

          \csname Folio@Le@\theFolioListNo\endcsname. Put a .pagebreak block after
          that exhibit in the manuscript, or move the exhibit}%
      \fi
    \fi}{\Folio@keeptrue}%
