  \par\addvspace{10pt}\noindent\begin{minipage}{\linewidth}
  {\color{FolioAccent}\hrule height 1pt}\vspace{9pt}
  \noindent\begin{minipage}[c]{.22\linewidth}
    \sbox{\FolioMetricBox}{{\color{FolioAccent}\sffamily\bfseries\fontsize{42}{43}\selectfont #1}}%
    \ifdim\wd\FolioMetricBox>\linewidth
      \resizebox{\linewidth}{!}{\usebox{\FolioMetricBox}}%
    \else\usebox{\FolioMetricBox}\fi
  \end{minipage}\hfill\begin{minipage}[c]{.74\linewidth}\RaggedRight
}{\end{minipage}\par\vspace{9pt}{\color{FolioRule}\hrule}\end{minipage}\par\addvspace{5pt}}
