% An editorial report with text columns, full-width evidence and a quiet footer.
\usepackage{geometry}
\usepackage{microtype}
\geometry{paperwidth=612bp,paperheight=792bp,left=80bp,right=50bp,top=64bp,bottom=74bp,footskip=30bp}
\definecolor{FolioAccent}{HTML}{2E559C}
\definecolor{FolioInk}{HTML}{1A2636}
\definecolor{FolioMuted}{HTML}{46566D}
\definecolor{FolioRule}{HTML}{A8B5C7}
\newcommand{\FolioKind}{Outlook report}
\renewcommand{\familydefault}{\sfdefault}
\renewcommand{\maketitle}{\FolioJournalTitle{47}{48}}
\AtBeginDocument{%
  % Justified, hyphenated text; microtype lets punctuation hang in the margin.
  \fontsize{11}{15.5}\selectfont
  \tolerance=1800 \emergencystretch=2.5em \hyphenpenalty=10 \exhyphenpenalty=10
  \lefthyphenmin=2 \righthyphenmin=3
  % Word spaces may close up a little more than the font allows by default: in a narrow
  % column a slightly tight line reads better than a gappy one.
  \spaceskip=\fontdimen2\font plus \fontdimen3\font minus 1.5\fontdimen4\font
  % Lines looser than this are reported by the build for the author to reword.
  \hbadness=4000
  \setlength{\parskip}{7pt plus 1.5pt minus 1pt}%
  \setlength{\skip\footins}{23pt plus 3pt minus 3pt}}
\def\footnoterule{%
  \kern-3pt{\color{FolioRule}\hrule width\textwidth height .45pt}\kern6pt}
\titleformat{\section}{\FolioDisplay\fontsize{31}{33}\selectfont\color{FolioInk}}%
  {\thesection}{.6em}{\FolioTitleRaggedRight}
\titlespacing*{\section}{0pt}{30pt}{14pt}
% A section starts on a new page when less than this share of the page is left.
\newcommand{\FolioSectionMinPage}{.33}
\newcommand{\FolioBeforeSection}{\Needspace{\FolioSectionMinPage\textheight}}
\titleformat{\subsection}{\FolioDisplay\fontsize{16}{19}\selectfont\color{FolioInk}}%
  {\thesubsection}{.65em}{\raggedright}
\titlespacing*{\subsection}{0pt}{20pt}{6pt}
